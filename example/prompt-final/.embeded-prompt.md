# Apple Freefall Simulation System Documentation

## 1. Introduction
### 1.1 Purpose
The Apple Freefall Simulation is a Python application that models the freefall motion of an apple under Earth's gravity. The system allows users to input physical parameters and visualize the apple's trajectory over time, demonstrating basic principles of kinematics.

### 1.2 Goals
- Provide an educational tool for understanding freefall motion
- Allow users to experiment with different physical parameters
- Generate clear output of the simulation results
- Implement a simple configuration system for simulation parameters

### 1.3 Scope
The system will:
- Accept user input for physical parameters (initial height, mass, etc.)
- Read simulation parameters from a config.json file
- Calculate the apple's position over time during freefall
- Output the evaluation results to out.md
- Handle basic error cases for invalid inputs

## 5. Mathematical Model

### 5.1 Basic Freefall (No Air Resistance)
Position as a function of time:

y(t) = y_0 - 0.5 * g * t^2

Velocity as a function of time:

v(t) = -g * t

Where:
- y(t) is the height at time t
- y_0 is the initial height
- g is the acceleration due to gravity (9.81 m/s²)
- v(t) is the velocity at time t

### 5.2 Freefall with Air Resistance
Air resistance force:

F_air = 0.5 * ρ * C_d * A * v^2

Where:
- ρ is the air density
- C_d is the drag coefficient
- A is the cross-sectional area
- v is the velocity

Net acceleration:

a = g - (F_air / m)

Where:
- m is the mass of the apple

class ObjectManager:
    def __init__(self):
        self.objects = []
        self.next_id = 1

    def add_object(self, obj):
        obj.id = self.next_id
        self.next_id += 1
        self.objects.append(obj)
        return obj.id

    def remove_object(self, object_id):
        self.objects = [obj for obj in self.objects if obj.id != object_id]

    def get_object(self, object_id):
        for obj in self.objects:
            if obj.id == object_id:
                return obj
        return None

    def clear_objects(self):
        self.objects = []
        self.next_id = 1

    def update_object_position(self, object_id, x, y):
        obj = self.get_object(object_id)
        if obj:
            obj.x = x
            obj.y = y
            return True
        return False

import numpy as np

class PhysicsEngine:
    def __init__(self):
        self.G = 100.0  # Gravitational constant (simplified for visualization)
        self.epsilon = 1.0  # Small constant to prevent division by zero

    def update(self, objects, simulation_speed):
        # Update positions based on velocities
        for obj in objects:
            obj.x += obj.velocity[0] * simulation_speed * 0.01
            obj.y += obj.velocity[1] * simulation_speed * 0.01

        # Update velocities based on gravitational forces
        for i, obj1 in enumerate(objects):
            net_force = np.array([0.0, 0.0])

            for j, obj2 in enumerate(objects):
                if i != j:
                    force = self._calculate_gravitational_force(obj1, obj2)
                    net_force += force

            # Update velocity (F = ma, so a = F/m)
            acceleration = net_force / obj1.mass
            obj1.velocity += acceleration * simulation_speed * 0.01

    def _calculate_gravitational_force(self, obj1, obj2):
        dx = obj2.x - obj1.x
        dy = obj2.y - obj1.y
        distance_squared = dx**2 + dy**2 + self.epsilon
        distance = np.sqrt(distance_squared)

        force_magnitude = (self.G * obj1.mass * obj2.mass) / distance_squared
        force_direction = np.array([dx, dy]) / distance

        return force_magnitude * force_direction

    def calculate_curvature(self, x, y, objects):
        curvature = 0.0
        for obj in objects:
            dx = x - obj.x
            dy = y - obj.y
            distance_squared = dx**2 + dy**2 + self.epsilon
            curvature += (self.G * obj.mass) / distance_squared
        return curvature

from PyQt5.QtCore import QObject, pyqtSignal, QTimer
from simulation.engine.object_manager import ObjectManager
from simulation.engine.physics_engine import PhysicsEngine
from simulation.models.celestial_object import CelestialObject
import numpy as np

class SimulationEngine(QObject):
    simulation_updated = pyqtSignal(object)  # Emits simulation state
    object_selected = pyqtSignal(object)     # Emits selected object

    def __init__(self):
        super().__init__()
        self.object_manager = ObjectManager()
        self.physics_engine = PhysicsEngine()
        self.simulation_timer = QTimer()
        self.simulation_timer.timeout.connect(self.update_simulation)
        self.simulation_speed = 1.0
        self.is_running = False
        self.selected_object = None

        # Initialize with some default objects
        self._initialize_default_objects()

    def _initialize_default_objects(self):
        # Add a central black hole
        black_hole = CelestialObject(
            x=400, y=300, mass=1000, radius=20,
            color=(0, 0, 0), name="Black Hole"
        )
        self.object_manager.add_object(black_hole)

        # Add a planet
        planet = CelestialObject(
            x=600, y=300, mass=10, radius=10,
            color=(0, 0, 255), name="Planet"
        )
        planet.velocity = np.array([0, 50])
        self.object_manager.add_object(planet)

    def start(self):
        if not self.is_running:
            self.simulation_timer.start(16)  # ~60 FPS
            self.is_running = True

    def pause(self):
        if self.is_running:
            self.simulation_timer.stop()
            self.is_running = False

    def reset(self):
        self.pause()
        self.object_manager.clear_objects()
        self._initialize_default_objects()
        self.selected_object = None
        self.simulation_updated.emit(self.get_simulation_state())

    def update_simulation(self):
        # Update physics
        self.physics_engine.update(self.object_manager.objects, self.simulation_speed)

        # Emit updated state
        self.simulation_updated.emit(self.get_simulation_state())

    def get_simulation_state(self):
        return {
            'objects': self.object_manager.objects,
            'selected_object': self.selected_object
        }

    def set_simulation_speed(self, speed):
        self.simulation_speed = speed

    def add_object(self, x, y):
        new_object = CelestialObject(
            x=x, y=y, mass=10, radius=10,
            color=(255, 0, 0), name=f"Object {len(self.object_manager.objects) + 1}"
        )
        self.object_manager.add_object(new_object)
        return new_object

    def select_object(self, object_id):
        self.selected_object = self.object_manager.get_object(object_id)
        self.object_selected.emit(self.selected_object)

    def update_selected_object_mass(self, mass):
        if self.selected_object:
            self.selected_object.mass = mass
            self.simulation_updated.emit(self.get_simulation_state())

    def update_selected_object_color(self, color):
        if self.selected_object:
            self.selected_object.color = color
            self.simulation_updated.emit(self.get_simulation_state())

    def get_position(self):

    def set_position(self, x, y):
        self.x = x
        self.y = y
        self.update_trail()

from PyQt5.QtWidgets import QWidget
from PyQt5.QtGui import QPainter, QColor, QPen, QBrush, QFont
from PyQt5.QtCore import Qt, QPoint, QTimer
import numpy as np

class SimulationCanvas(QWidget):
    def __init__(self, simulation_engine):
        super().__init__()
        self.simulation_engine = simulation_engine
        self.grid_visible = True
        self.trail_visible = True
        self.simulation_state = None
        self.setMinimumSize(600, 600)

        # Mouse interaction
        self.dragging_object = None
        self.drag_offset = None

        # Connect to simulation updates
        self.simulation_engine.simulation_updated.connect(self.update_simulation_state)

    def update_simulation_state(self, state):
        self.simulation_state = state
        self.update()  # Trigger repaint

    def set_grid_visible(self, visible):
        self.grid_visible = visible
        self.update()

    def set_trail_visible(self, visible):
        self.trail_visible = visible
        self.update()

    def reset(self):
        self.dragging_object = None
        self.drag_offset = None
        self.update()

    def paintEvent(self, event):
        if not self.simulation_state:
            return

        painter = QPainter(self)
        painter.setRenderHint(QPainter.Antialiasing)

        # Draw background
        painter.fillRect(self.rect(), Qt.white)

        # Draw grid if visible
        if self.grid_visible:
            self._draw_grid(painter)

        # Draw spacetime curvature
        self._draw_curvature(painter)

        # Draw objects
        for obj in self.simulation_state['objects']:
            self._draw_object(painter, obj)

        # Draw trails if visible
        if self.trail_visible:
            for obj in self.simulation_state['objects']:
                self._draw_trail(painter, obj)

        # Draw selected object highlight
        if self.simulation_state['selected_object']:
            self._draw_selection_highlight(painter, self.simulation_state['selected_object'])

    def _draw_grid(self, painter):
        pen = QPen(QColor(200, 200, 200))
        pen.setStyle(Qt.DotLine)
        painter.setPen(pen)

        width = self.width()
        height = self.height()

        # Draw vertical lines
        for x in range(0, width, 20):
            painter.drawLine(x, 0, x, height)

        # Draw horizontal lines
        for y in range(0, height, 20):
            painter.drawLine(0, y, width, y)

    def _draw_curvature(self, painter):
        width = self.width()
        height = self.height()
        grid_size = 20

        # Create a gradient based on curvature
        for x in range(0, width, grid_size):
            for y in range(0, height, grid_size):
                curvature = self.simulation_engine.physics_engine.calculate_curvature(
                    x, y, self.simulation_state['objects']
                )

                # Normalize curvature for visualization
                normalized = min(curvature / 1000, 1.0)
                color_value = int(255 * (1 - normalized))
                color = QColor(color_value, color_value, 255)

                painter.fillRect(x, y, grid_size, grid_size, color)

    def _draw_object(self, painter, obj):
        # Draw object
        painter.setBrush(QBrush(QColor(*obj.color)))
        painter.setPen(Qt.NoPen)
        painter.drawEllipse(int(obj.x - obj.radius), int(obj.y - obj.radius),
                           int(obj.radius * 2), int(obj.radius * 2))

        # Draw object name
        painter.setPen(QPen(Qt.black))
        painter.setFont(QFont("Arial", 8))
        painter.drawText(int(obj.x + obj.radius + 5), int(obj.y - obj.radius - 5), obj.name)

    def _draw_trail(self, painter, obj):
        if len(obj.trail) < 2:
            return

        pen = QPen(QColor(*obj.color))
        pen.setWidth(2)
        painter.setPen(pen)

        for i in range(1, len(obj.trail)):
            start = obj.trail[i-1]
            end = obj.trail[i]
            painter.drawLine(int(start[0]), int(start[1]), int(end[0]), int(end[1]))

    def _draw_selection_highlight(self, painter, obj):
        pen = QPen(QColor(255, 255, 0))

    def mousePressEvent(self, event):

            return

        pos = event.pos()

        # Check if we clicked on an object
        for obj in self.simulation_state['objects']:
            dx = pos.x() - obj.x
            dy = pos.y() - obj.y
            distance = (dx**2 + dy**2)**0.5

            if distance <= obj.radius:
                self.dragging_object = obj
                self.drag_offset = (dx, dy)
                self.simulation_engine.select_object(obj.id)
                return

        # If we didn't click on an object, create a new one
        self.simulation_engine.add_object(pos.x(), pos.y())

    def mouseMoveEvent(self, event):
        if self.dragging_object:
            new_x = event.pos().x() - self.drag_offset[0]
            new_y = event.pos().y() - self.drag_offset[1]
            self.simulation_engine.object_manager.update_object_position(
                self.dragging_object.id, new_x, new_y
            )

    def mouseReleaseEvent(self, event):
        self.dragging_object = None
        self.drag_offset = None
