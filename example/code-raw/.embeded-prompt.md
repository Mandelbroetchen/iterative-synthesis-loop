"""
Animation Engine Module
=======================

This module handles the real-time animation of the freefall simulation using Pygame.
"""

import pygame
import sys
import time
from typing import Tuple, Optional
from freefall_simulator import SimulationResult

class AnimationEngine:
    """Handles the real-time animation of the freefall simulation."""

    def __init__(self, result: SimulationResult, speed: float = 1.0):
        """
        Initialize the animation engine.

        Args:
            result: SimulationResult object containing simulation data
            speed: Animation speed multiplier
        """
        self.result = result
        self.speed = speed
        self.paused = False
        self.current_frame = 0
        self.last_frame_time = 0
        self.frame_delay = 0.033  # ~30 FPS

        # Initialize pygame
        pygame.init()

        # Screen dimensions
        self.screen_width = 800
        self.screen_height = 600
        self.screen = pygame.display.set_mode((self.screen_width, self.screen_height))
        pygame.display.set_caption("Freefall Simulation")

        # Colors
        self.colors = {
            'sky': (135, 206, 235),
            'ground': (139, 69, 19),
            'grass': (34, 139, 34),
            'apple': (255, 0, 0),
            'stem': (139, 69, 19),
            'text': (0, 0, 0),
            'button': (70, 130, 180),
            'button_hover': (100, 149, 237),
            'button_text': (255, 255, 255)
        }

        # Fonts
        self.font = pygame.font.SysFont('Arial', 18)
        self.title_font = pygame.font.SysFont('Arial', 24)

        # Apple properties
        self.apple_radius = 20
        self.ground_level = self.screen_height - 100

        # Calculate scaling factors
        self.max_height = max(self.result.position) if self.result.position else 1
        self.height_scale = (self.ground_level - 50) / self.max_height if self.max_height > 0 else 1

        # Buttons
        self.play_button = pygame.Rect(50, self.screen_height - 50, 100, 40)
        self.restart_button = pygame.Rect(170, self.screen_height - 50, 100, 40)
        self.speed_slider = pygame.Rect(300, self.screen_height - 45, 200, 30)
        self.speed_knob = pygame.Rect(300, self.screen_height - 45, 15, 30)

        # Calculate knob position based on speed
        self.update_speed_knob()

    def update_speed_knob(self):
        """Update the speed knob position based on current speed."""
        knob_x = self.speed_slider.x + (self.speed - 0.1) / 4.9 * self.speed_slider.width
        knob_x = max(self.speed_slider.x, min(knob_x, self.speed_slider.x + self.speed_slider.width - 15))
        self.speed_knob.x = knob_x

    def draw_apple(self, position: float):
        """Draw the apple at the specified position."""
        # Calculate y position (invert since screen coordinates go top to bottom)
        y_pos = self.ground_level - position * self.height_scale

        # Ensure apple doesn't go below ground
        y_pos = max(self.ground_level - self.apple_radius, y_pos)

        # Draw apple
        pygame.draw.circle(self.screen, self.colors['apple'],
                          (self.screen_width // 2, int(y_pos)), self.apple_radius)

        # Draw stem
        stem_start = (self.screen_width // 2, int(y_pos) - self.apple_radius)
        stem_end = (self.screen_width // 2, int(y_pos) - self.apple_radius - 10)
        pygame.draw.line(self.screen, self.colors['stem'], stem_start, stem_end, 3)

    def draw_ground(self):
        """Draw the ground and grass."""
        # Ground
        pygame.draw.rect(self.screen, self.colors['ground'],
                        (0, self.ground_level, self.screen_width, self.screen_height - self.ground_level))

        # Grass
        for i in range(0, self.screen_width, 10):
            grass_height = 10 + (i % 20)
            pygame.draw.line(self.screen, self.colors['grass'],
                            (i, self.ground_level),
                            (i, self.ground_level - grass_height), 2)

    def draw_info_panel(self, time: float, position: float, velocity: float, acceleration: float):
        """Draw the information panel with simulation data."""
        info_y = self.ground_level + 20

        # Title
        title = self.title_font.render("Freefall Simulation", True, self.colors['text'])
        self.screen.blit(title, (self.screen_width // 2 - title.get_width() // 2, info_y))

        info_y += 40

        # Simulation data
        data = [
            f"Time: {time:.2f} s",
            f"Position: {max(position, 0.0):.2f} m",
            f"Velocity: {velocity:.2f} m/s",
            f"Acceleration: {acceleration:.2f} m/s²"
        ]

        for line in data:
            text = self.font.render(line, True, self.colors['text'])
            self.screen.blit(text, (50, info_y))
            info_y += 30

        # Warning if impact occurred
        if self.result.impact_time is not None:
            warning = self.font.render(f"Impact at: {self.result.impact_time:.2f} s", True, (255, 0, 0))
            self.screen.blit(warning, (50, info_y + 20))

    def draw_buttons(self):
        """Draw the control buttons."""
        mouse_pos = pygame.mouse.get_pos()

        # Play/Pause button
        play_color = self.colors['button_hover'] if self.play_button.collidepoint(mouse_pos) else self.colors['button']
        pygame.draw.rect(self.screen, play_color, self.play_button)
        pygame.draw.rect(self.screen, (0, 0, 0), self.play_button, 2)

        play_text = self.font.render("Pause" if not self.paused else "Play", True, self.colors['button_text'])
        self.screen.blit(play_text, (self.play_button.x + 25, self.play_button.y + 10))

        # Restart button
        restart_color = self.colors['button_hover'] if self.restart_button.collidepoint(mouse_pos) else self.colors['button']
        pygame.draw.rect(self.screen, restart_color, self.restart_button)
        pygame.draw.rect(self.screen, (0, 0, 0), self.restart_button, 2)

        restart_text = self.font.render("Restart", True, self.colors['button_text'])
        self.screen.blit(restart_text, (self.restart_button.x + 20, self.restart_button.y + 10))

        # Speed slider
        pygame.draw.rect(self.screen, (200, 200, 200), self.speed_slider)
        pygame.draw.rect(self.screen, (0, 0, 0), self.speed_slider, 2)

        # Speed knob
        pygame.draw.rect(self.screen, self.colors['button'], self.speed_knob)
        pygame.draw.rect(self.screen, (0, 0, 0), self.speed_knob, 2)

        speed_text = self.font.render(f"Speed: {self.speed:.1f}x", True, self.colors['text'])
        self.screen.blit(speed_text, (self.speed_slider.x + self.speed_slider.width + 10, self.speed_slider.y))

    def handle_events(self):
        """Handle pygame events."""
        for event in pygame.event.get():
            if event.type == pygame.QUIT:
                return False

            if event.type == pygame.MOUSEBUTTONDOWN:
                mouse_pos = pygame.mouse.get_pos()

                # Check button clicks
                if self.play_button.collidepoint(mouse_pos):
                    self.paused = not self.paused
                elif self.restart_button.collidepoint(mouse_pos):
                    self.current_frame = 0
                    self.paused = False
                elif self.speed_knob.collidepoint(mouse_pos):
                    # Start dragging speed knob
                    self.dragging = True
                elif self.speed_slider.collidepoint(mouse_pos):
                    # Jump to clicked position
                    self.speed = 0.1 + (mouse_pos[0] - self.speed_slider.x) / self.speed_slider.width * 4.9
                    self.speed = max(0.1, min(5.0, self.speed))
                    self.update_speed_knob()

            if event.type == pygame.MOUSEBUTTONUP:
                self.dragging = False

            if event.type == pygame.MOUSEMOTION and hasattr(self, 'dragging') and self.dragging:
                # Update speed based on knob position
                new_x = max(self.speed_slider.x, min(mouse_pos[0], self.speed_slider.x + self.speed_slider.width - 15))
                self.speed = 0.1 + (new_x - self.speed_slider.x) / self.speed_slider.width * 4.9
                self.speed_knob.x = new_x

        return True

    def render_frame(self):
        """Render a single frame of the animation."""
        # Clear screen
        self.screen.fill(self.colors['sky'])

        # Draw ground
        self.draw_ground()

        # Get current simulation data
        if self.current_frame < len(self.result.time):
            current_time = self.result.time[self.current_frame]
            current_position = self.result.position[self.current_frame]
            current_velocity = self.result.velocity[self.current_frame]
            current_acceleration = self.result.acceleration[self.current_frame]
        else:
            # If we've reached the end, use the last frame
            current_time = self.result.time[-1] if self.result.time else 0
            current_position = self.result.position[-1] if self.result.position else 0
            current_velocity = self.result.velocity[-1] if self.result.velocity else 0
            current_acceleration = self.result.acceleration[-1] if self.result.acceleration else 0

        # Draw apple
        self.draw_apple(current_position)

        # Draw info panel
        self.draw_info_panel(current_time, current_position, current_velocity, current_acceleration)

        # Draw buttons
        self.draw_buttons()

        # Update display
        pygame.display.flip()

    def run_animation(self):
        """Run the animation loop."""
        clock = pygame.time.Clock()
        running = True

        while running:
            # Handle events
            running = self.handle_events()

            # Update animation if not paused
            if not self.paused and running:
                current_time = time.time()
                if current_time - self.last_frame_time >= self.frame_delay / self.speed:
                    if self.current_frame < len(self.result.time) - 1:
                        self.current_frame += 1
                    self.last_frame_time = current_time

            # Render frame
            self.render_frame()

            # Cap the frame rate
            clock.tick(60)

        pygame.quit()
        sys.exit()

"""
Freefall Simulation Module
==========================

This module implements the physics calculations and simulation logic for freefall motion
under constant gravitational acceleration.
"""

from typing import List, Optional, Tuple
import math

class SimulationResult:
    """Container for simulation results at each time step."""

    def __init__(self, time: List[float], position: List[float], velocity: List[float],
                 acceleration: List[float], impact_time: Optional[float] = None):
        """
        Initialize simulation results.

        Args:
            time: List of time values
            position: List of position values
            velocity: List of velocity values
            acceleration: List of acceleration values
            impact_time: Time when ground collision occurred (None if no collision)
        """
        self.time = time
        self.position = position
        self.velocity = velocity
        self.acceleration = acceleration
        self.impact_time = impact_time
        self.completed = impact_time is None

    def display(self) -> str:
        """Format simulation results as a table string."""
        header = "Time (s) | Position (m) | Velocity (m/s) | Acceleration (m/s²)"
        separator = "-" * len(header)
        rows = []
        for t, h, v, a in zip(self.time, self.position, self.velocity, self.acceleration):
            if h <= 0 and len(rows) > 0:
                break
            rows.append(f"{t:8.2f} | {max(h, 0.0):12.2f} | {v:14.2f} | {a:18.2f}")
        return "\n".join([header, separator] + rows)

    def get_summary(self) -> str:
        """Get a summary of simulation parameters and results."""
        summary = [
            "Simulation Parameters:",
            f"- Initial height: {self.position[0] if self.position else 0:.2f} m",
            f"- Initial velocity: {self.velocity[0] if self.velocity else 0:.2f} m/s",
            f"- Time step: {self.time[1] - self.time[0] if len(self.time) > 1 else 0:.2f} s",
            f"- Gravitational acceleration: {self.acceleration[0] if self.acceleration else 0:.2f} m/s²"
        ]
        if self.impact_time is not None:
            summary.append(f"- Ground collision at: {self.impact_time:.2f} s")
        return "\n".join(summary)

    def get_impact_info(self) -> Tuple[float, float]:
        """
        Get impact information if collision occurred.

        Returns:
            Tuple of (impact_time, impact_velocity) or (None, None) if no collision
        """
        if self.impact_time is not None:
            idx = self.time.index(self.impact_time)
            return (self.impact_time, self.velocity[idx])
        return (None, None)

class InputValidator:
    """Handles validation of input parameters for the simulation."""

    @staticmethod
    def validate_height(h0: float) -> bool:
        """Validate initial height."""
        return h0 >= 0

    @staticmethod
    def validate_velocity(v0: float) -> bool:
        """Validate initial velocity."""
        return True  # Velocity can be any real number

    @staticmethod
    def validate_time_step(dt: float) -> bool:
        """Validate time step."""
        return dt > 0

    @staticmethod
    def validate_duration(T: float) -> bool:
        """Validate total duration."""
        return T > 0

    @staticmethod
    def validate_gravity(g: float) -> bool:
        """Validate gravitational acceleration."""
        return g >= 0

    @staticmethod
    def validate_animation_speed(speed: float) -> bool:
        """Validate animation speed."""
        return speed > 0

    @staticmethod
    def validate_all(h0: float, v0: float, dt: float, T: float, g: float, speed: float) -> bool:
        """Validate all input parameters."""
        return (InputValidator.validate_height(h0) and
                InputValidator.validate_velocity(v0) and
                InputValidator.validate_time_step(dt) and
                InputValidator.validate_duration(T) and
                InputValidator.validate_gravity(g) and
                InputValidator.validate_animation_speed(speed))

class WarningGenerator:
    """Generates warning messages for simulation events."""

    @staticmethod
    def ground_collision_warning(impact_time: float) -> str:
        """Generate warning for ground collision."""
        return f"Warning: Simulation stopped early - apple hit the ground at t ≈ {impact_time:.2f} s"

    @staticmethod
    def early_termination_warning() -> str:
        """Generate warning for early termination."""
        return "Warning: Simulation terminated early due to ground collision"

class FreefallSimulator:
    """Core physics engine for freefall simulation."""

    def __init__(self, h0: float, v0: float, dt: float, T: float, g: float = 9.81):
        """
        Initialize the freefall simulator with initial conditions.

        Args:
            h0: Initial height (meters)
            v0: Initial velocity (meters/second)
            dt: Time step (seconds)
            T: Total simulation duration (seconds)
            g: Gravitational acceleration (meters/second²), default 9.81

        Raises:
            ValueError: If input parameters are invalid
        """
        if not InputValidator.validate_all(h0, v0, dt, T, g, 1.0):
            raise ValueError("Invalid input parameters")

        self.h0 = h0
        self.v0 = v0
        self.dt = dt
        self.T = T
        self.g = g

    def calculate_position(self, t: float) -> float:
        """Calculate position at time t using kinematic equation."""
        return self.h0 + self.v0 * t + 0.5 * self.g * t ** 2

    def calculate_velocity(self, t: float) -> float:
        """Calculate velocity at time t using kinematic equation."""
        return self.v0 + self.g * t

    def detect_ground_collision(self, t: float, h: float) -> bool:
        """Check if ground collision has occurred."""
        return h <= 0

    def calculate_impact_time(self) -> Optional[float]:
        """
        Calculate exact time of ground impact using quadratic formula.

        Returns:
            Time of impact or None if no impact occurs
        """
        # Solve h(t) = 0: h0 + v0*t + 0.5*g*t² = 0
        discriminant = self.v0 ** 2 + 2 * self.g * self.h0
        if discriminant < 0:
            return None  # No real solution (apple never hits ground)

        t1 = (-self.v0 + math.sqrt(discriminant)) / self.g
        t2 = (-self.v0 - math.sqrt(discriminant)) / self.g

        # Return the positive time that's within simulation duration
        for t in [t1, t2]:
            if t >= 0 and t <= self.T:
                return t
        return None

    def run_simulation(self) -> SimulationResult:
        """Run the simulation and return results at each time step."""
        time_points = []
        position_values = []
        velocity_values = []
        acceleration_values = []
        impact_time = None

        current_time = 0.0
        while current_time <= self.T + 1e-9:  # Account for floating point precision
            h = self.calculate_position(current_time)
            v = self.calculate_velocity(current_time)

            time_points.append(current_time)
            position_values.append(h)
            velocity_values.append(v)
            acceleration_values.append(self.g)

            if self.detect_ground_collision(current_time, h) and impact_time is None:
                impact_time = current_time
                break

            current_time += self.dt

        # If we didn't detect impact during time steps, calculate exact impact time
        if impact_time is None:
            exact_impact = self.calculate_impact_time()
            if exact_impact is not None:
                impact_time = exact_impact

        return SimulationResult(time_points, position_values, velocity_values,
                              acceleration_values, impact_time)

from freefall_simulator import InputValidator

def get_float_input(prompt: str, min_val: float = None, max_val: float = None, default: float = None) -> float:
    """
    Get and validate a float input from the user.

    Args:
        prompt: Input prompt message
        min_val: Minimum allowed value (inclusive)
        max_val: Maximum allowed value (inclusive)
        default: Default value if user enters nothing

    Returns:
        Validated float value

    Raises:
        ValueError: If input cannot be converted to float or is out of bounds
    """
    while True:
        try:
            value = input(prompt)
            if value == "" and default is not None:
                return default
            value = float(value)
            if min_val is not None and value < min_val:
                print(f"Value must be ≥ {min_val}")
                continue
            if max_val is not None and value > max_val:
                print(f"Value must be ≤ {max_val}")
                continue
            return value
        except ValueError:
            print("Please enter a valid number")

def collect_parameters() -> dict:
    """
    Collect all simulation parameters from user input.

    Returns:
        Dictionary containing all simulation parameters
    """
    print("Freefall Simulation - Parameter Input")
    print("------------------------------------")

    params = {
        'h0': get_float_input("Initial height (m): ", min_val=0),
        'v0': get_float_input("Initial velocity (m/s): ", default=0),
        'dt': get_float_input("Time step (s): ", min_val=0.001, max_val=10, default=0.05),
        'T': get_float_input("Total duration (s): ", min_val=0.001, max_val=3600, default=5.0),
        'g': get_float_input("Gravitational acceleration (m/s²) [default=9.81]: ", min_val=0, default=9.81),
        'speed': get_float_input("Animation speed [default=1.0]: ", min_val=0.1, max_val=5.0, default=1.0)
    }

    return params

"""
Simulation Controller Module
============================

This module manages the overall simulation flow and coordinates between components.
"""

from freefall_simulator import FreefallSimulator, SimulationResult
from input_handler import collect_parameters
from animation_engine import AnimationEngine

class SimulationController:
    """Manages the overall simulation flow."""

    def __init__(self):
        """Initialize the simulation controller."""
        self.simulator = None
        self.result = None
        self.animation_engine = None

    def run_simulation(self):
        """Run the complete simulation workflow."""
        try:
            # Step 1: Collect input parameters
            params = collect_parameters()

            # Step 2: Initialize and run physics simulation
            self.simulator = FreefallSimulator(
                h0=params['h0'],
                v0=params['v0'],
                dt=params['dt'],
                T=params['T'],
                g=params['g']
            )
            self.result = self.simulator.run_simulation()

            # Step 3: Initialize and run animation
            self.animation_engine = AnimationEngine(self.result, params['speed'])
            self.animation_engine.run_animation()

        except ValueError as ve:
            print(f"Input error: {ve}")
        except Exception as e:
            print(f"An unexpected error occurred: {e}")
            print("Please try again with valid inputs.")

def main():
    """Main entry point for the simulation."""
    controller = SimulationController()
    controller.run_simulation()

if __name__ == "__main__":
    main()
